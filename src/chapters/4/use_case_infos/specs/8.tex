\subsection{Đặc tả Use Case ``Gửi tin nhắn''}

\subsubsection{Tóm tắt}
Người dùng gửi tin nhắn (Người dùng bao gồm bác sĩ và bệnh nhân).

\subsubsection{Dòng sự kiện}
\paragraph{\textbf{Dòng sự kiện chính:}}
\begin{enumerate}
  \item Người dùng đang ở giao diện soạn thảo tin nhắn với tin nhắn đã soạn.
  \item Người dùng nhấn nút ``Send''.
  \item Hệ thống kiểm tra, nếu hợp lệ thì gửi tin nhắn đến người nhận và hiển thị tình trạng xem của tin nhắn vừa gửi (``Not seen'', ``Seen'').
\end{enumerate}

\paragraph{\textbf{Các dòng sự kiện khác:}}
  \begin{itemize}
    \item Nếu người dùng không nhập tin nhắn mà nhấn nút ``Gửi'', hệ thống không gửi tin nhắn.
    \item Nếu người dụng nhấn vào tin nhắn bất kì do mỉnh gửi, hệ thống hiển thị trạng thái xem của tin nhắn đó.
  \end{itemize}

\subsubsection{Các yêu cầu đặc biệt}
\begin{itemize}
  \item Nội dung tin nhắn không được để trống.
  \item Xảy ra lỗi nếu: Không nhập nội dung tin nhắn.
\end{itemize}

\subsubsection{Trạng thái hệ thống khi bắt đầu thực hiện Use Case}
Người dùng đang ở giao diện soạn thảo tin nhắn.

\subsubsection{Trạng thái hệ thống sau khi thực hiện Use Case}
\begin{itemize}
  \item Thành công: Tin nhắn của người dùng được gửi đến người nhận.
  \item Thất bại: Thông báo.
\end{itemize}
