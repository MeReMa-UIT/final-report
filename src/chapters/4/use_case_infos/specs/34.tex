\subsection{Đặc tả Use Case ``Lập báo cáo thống kê số lượng bệnh án tiếp nhận''}

\subsubsection{Tóm tắt}
Admin tạo báo cáo thống kê số lượng bệnh án tiếp nhận.

\subsubsection{Dòng sự kiện}
\paragraph{\textbf{Dòng sự kiện chính:}}
\begin{enumerate}
  \item Admin đang ở giao diện chính của hệ thống.
  \item Admin nhấn nút ``Statistics'', hệ thống hiển thị giao diện quản lí báo cáo.
  \item Admin nhập các thông tin cần thiết.
  \item Hệ thống kiểm tra và tạo báo cáo theo yêu cầu.
\end{enumerate}

\paragraph{\textbf{Các dòng sự kiện khác:}}
\begin{enumerate}
  \item Nội dung báo cáo:
    \begin{itemize}
      \item Thời gian.
      \item Số lượng bệnh án tiếp nhận.
      \item Số lượng bệnh án theo bác sĩ.
      \item Số lượng bệnh án theo chẩn đoán
    \end{itemize}
\end{enumerate}

\subsubsection{Các yêu cầu đặc biệt}
\begin{itemize}
  \item Thông tin nhập vào là thời gian và khoảng thời gian.
  \item Hệ thống phải tự động tạo báo cáo theo yêu cầu của admin.
\end{itemize}

\subsubsection{Trạng thái hệ thống khi bắt đầu thực hiện Use Case}
Admin đang ở giao diện chính của hệ thống.

\subsubsection{Trạng thái hệ thống sau khi thực hiện Use Case}
Hệ thống đã tạo báo cáo thành công và hiển thị.
