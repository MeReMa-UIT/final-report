\subsection{Đặc tả Use Case ``Lập đơn thuốc''}

\subsubsection{Tóm tắt}
Bác sĩ lập đơn thuốc cho bệnh nhân.

\subsubsection{Dòng sự kiện}
\paragraph{\textbf{Dòng sự kiện chính}}
\begin{enumerate}
  \item Bác sĩ đang ở giao diện xem chi tiết bệnh án.
  \item Bác sĩ chuyển sang tab ``Prescriptions''.
  \item Bác sĩ nhấn nút ``Create Prescription'', hệ thống chuyển sang giao diện tạo đơn thuốc.
  \item Bác sĩ nhập thông tin đơn thuốc.
  \item Bác sĩ nhấn nút lưu đơn thuốc, hệ thống kiểm tra thông tin đơn thuốc và ghi nhận.
\end{enumerate}

\paragraph{\textbf{Các dòng sự kiện khác}}
\begin{enumerate}
  \item Nếu bác sĩ nhấn nút ``X'', hệ thống quay về giao diện quản lí đơn thuốc.
  \item Nếu bác sĩ nhập đơn thuốc không hợp lệ, hệ thống thông báo lỗi và yêu cầu nhập lại.
\end{enumerate}

\subsubsection{Các yêu cầu đặc biệt}
\begin{itemize}
  \item Các thông tin nhập vào phải là kí tự Unicode.
  \item Các trường thông tin phải nhập đúng kiểu dữ liệu.
  \item Bác sĩ có thể chọn nhiều loại thuốc cho đơn thuốc.
  \item Mỗi bệnh án chỉ có thể có một đơn thuốc duy nhất.
  \item Các trường thông tin bao gồm:
  \begin{itemize}
    \item Thuốc bảo hiểm.
    \item Ghi chú.
    \item Thông tin các loại thuốc:
    \begin{itemize}
      \item Tên thuốc.
      \item Đơn vị tính.
      \item Thời gian sử dụng.
      \item Số lượng sử dụng.
      \item Tần suất sử dụng.
      \item Cách dùng.
    \end{itemize}
  \end{itemize}
  \item Yêu cầu về việc thông báo lỗi nhập sai dữ liệu:
  \begin{itemize}
    \item Màn hình hiển thị thông báo lỗi.
    \item Các trường dữ liệu không hợp lệ được bôi đỏ.
  \end{itemize}
  \item Xảy ra lỗi nếu:
  \begin{itemize}
    \item Nhập thiếu các trường thông tin bắt buộc.
    \item Nhập không đúng kí tự Unicode.
    \item Nhập thông tin không đúng kiểu dữ liệu.
  \end{itemize}
\end{itemize}

\subsubsection{Trạng thái hệ thống khi bắt đầu thực hiện Use Case}
Bác sĩ đang ở giao diện xem chi tiết bệnh án.

\subsubsection{Trạng thái hệ thống sau khi thực hiện Use Case}
\begin{itemize}
  \item Thành công: Hệ thống lưu lại thông tin đơn thuốc và chuyển về giao diện quản lí đơn thuốc.
  \item Không thành công: Thông báo.
\end{itemize}
