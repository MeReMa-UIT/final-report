\chapter{CÁC YÊU CẦU PHI CHỨC NĂNG ĐẶT RA CHO HỆ THỐNG}

\section{Yêu cầu về bảo mật}

\subsection{Xác thực và phân quyền}
\begin{itemize}
  \item Hệ thống phải hỗ trợ xác thực người dùng thông qua tên đăng nhập và mật khẩu
  \item Phân quyền truy cập dựa trên vai trò của người dùng (admin, người dùng thường)
  \item Mã hóa mật khẩu người dùng trước khi lưu trữ trong cơ sở dữ liệu
\end{itemize}

\subsection{Bảo vệ dữ liệu}
\begin{itemize}
  \item Mã hóa dữ liệu nhạy cảm trong quá trình truyền tải (HTTPS/SSL)
  \item Thực hiện backup dữ liệu định kỳ và bảo mật
  \item Kiểm soát truy cập vào dữ liệu theo nguyên tắc cần biết
\end{itemize}

\section{Yêu cầu về hiệu suất}

\subsection{Thời gian phản hồi}
\begin{itemize}
  \item Thời gian phản hồi của hệ thống không quá 3 giây cho các thao tác thông thường
  \item Thời gian tải trang không quá 5 giây trong điều kiện mạng bình thường
  \item Thời gian xử lý truy vấn cơ sở dữ liệu không quá 2 giây đối với các truy vấn phức tạp hay kết quả trả về lớn
\end{itemize}

\subsection{Khả năng mở rộng}
\begin{itemize}
  \item Hệ thống có thể hỗ trợ tối thiểu 100 người dùng đồng thời
  \item Khả năng mở rộng theo chiều ngang khi số lượng người dùng tăng
  \item Tối ưu hóa sử dụng tài nguyên hệ thống (CPU, RAM, storage)
\end{itemize}

\section{Yêu cầu về độ tin cậy}

\subsection{Tính khả dụng}
\begin{itemize}
  \item Hệ thống phải hoạt động 24/7 với uptime tối thiểu 99\%
  \item Thời gian ngừng hoạt động không được vượt quá 1\% tổng thời gian trong tháng
  \item Cung cấp cơ chế fallback khi có sự cố xảy ra
\end{itemize}

\subsection{Khôi phục sau sự cố}
\begin{itemize}
  \item Thời gian khôi phục hệ thống sau sự cố không quá 30 phút
  \item Tự động backup dữ liệu mỗi ngày và lưu trữ tối thiểu 30 ngày
\end{itemize}

\subsection{Xử lý lỗi}
\begin{itemize}
  \item Hệ thống phải xử lý graceful các lỗi không mong muốn
  \item Ghi log chi tiết các lỗi để hỗ trợ debugging
  \item Thông báo lỗi rõ ràng và hữu ích cho người dùng
\end{itemize}
